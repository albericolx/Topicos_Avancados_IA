\subsection{Representação Explícita em Grafo Direcionado (\texttt{nx.DiGraph})}
A classe \texttt{MLPGraph} (\texttt{code/mlp/core\_mlp/graph.py}) instancia cada neurônio, nó de \textit{bias} e sinapse como elementos explícitos de um grafo direcionado $G = (V, E)$, conforme ilustrado na Figura~\ref{fig:arquitetura_diagrama}.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.98\textwidth]{figs/arquitetura_diagrama.png}
    \caption{Desenho da arquitetura da MLP em grafo direcionado (\texttt{nx.DiGraph}) e fluxo matemático bidirecional (\textit{Forward Pass} e \textit{Backpropagation}).}
    \label{fig:arquitetura_diagrama}
\end{figure}

O funcionamento estrutural do grafo organiza-se em três componentes:
\begin{itemize}
    \item \textbf{Vértices Neurais (\texttt{L\{l\}\_N\{j\}}):} armazenam o potencial pós-sináptico \texttt{z} ($z_j^{(l)}$), a ativação \texttt{value} ($a_j^{(l)}$) e o erro retropropagado \texttt{delta} ($\delta_j^{(l)}$).
    \item \textbf{Vértices de Bias (\texttt{L\{l\}\_bias}):} cada camada $l \in \{0, \dots, L-2\}$ possui um nó de bias explícito com ativação constante $a_{\text{bias}}^{(l)} = +1$, conectado a todos os neurônios da camada $l+1$. Essa estrutura corrige a limitação da versão preliminar (\texttt{C0}), na qual a Camada Oculta 1 não possuía termo de bias.
    \item \textbf{Arestas Sinápticas (\texttt{(u, v)}):} armazenam \texttt{weight} ($w_{uv}$), \texttt{prev\_weight} ($w_{uv}^{(t-1)}$), \texttt{delta\_weight} ($\Delta w_{uv}$), \texttt{gradient} ($\partial \mathcal{L}/\partial w_{uv}$) e \texttt{velocity} ($v_{uv}$). Na topologia campeã \texttt{[13$\to$16$\to$8$\to$1]}, o grafo gerencia $(14\times 16) + (17\times 8) + (9\times 1) = 369$ conexões sinápticas.
\end{itemize}

\subsection{Propagação Direta (\textit{Forward Pass}) e Retropropagação (\textit{Backpropagation})}
No \textit{Forward Pass}, cada neurônio $j$ da camada $l = 1, \dots, L-1$ computa:
\begin{equation}
    z_j^{(l)} = \sum_{i=0}^{N_{l-1}-1} w_{ij}^{(l)} a_i^{(l-1)} + b_j^{(l)}, \qquad a_j^{(l)} = f^{(l)}\!\left(z_j^{(l)}\right)
\end{equation}
sendo a saída $\hat{y} = \sigma(z_{\text{out}})$. Na retropropagação sob perda \textit{Binary Cross-Entropy} (BCE) com regularização $L_2$:
\begin{equation}
    \mathcal{L}(\hat{y}, y) = -\Big[y \ln(\hat{y}) + (1 - y)\ln(1 - \hat{y})\Big] + \frac{\lambda}{2}\sum_{l,i,j}\left(w_{ij}^{(l)}\right)^2
\end{equation}
o termo $\hat{y}(1-\hat{y})$ da derivada da sigmoide cancela-se analiticamente com o denominador de $\partial \mathcal{L}/\partial \hat{y}$, resultando em:
\begin{equation}
    \delta^{(L-1)} = \frac{\partial \mathcal{L}}{\partial z_{\text{out}}} = \hat{y} - y, \qquad \delta_j^{(l)} = \left(\sum_{k} w_{jk}^{(l+1)} \delta_k^{(l+1)}\right) f'^{(l)}\!\left(z_j^{(l)}\right)
\end{equation}
Por fim, os pesos sinápticos são atualizados com \textit{Momentum} ($\beta$) e decaimento $L_2$ ($\lambda$):
\begin{equation}
    g_{ij}^{(l)} = a_i^{(l-1)}\delta_j^{(l)} + \lambda w_{ij}^{(l)}, \qquad v_{ij}^{(l)} \leftarrow \beta v_{ij}^{(l)} + g_{ij}^{(l)}, \qquad w_{ij}^{(l)} \leftarrow w_{ij}^{(l)} - \eta_t v_{ij}^{(l)}
\end{equation}
