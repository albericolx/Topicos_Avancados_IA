\subsection{Justificativa dos Parâmetros Modificados}
\begin{enumerate}
    \item \textbf{Derivada Analítica da BCE e Inclusão do Bias \texttt{L0\_bias} (C0 $\to$ C1):} A eliminação do fator espúrio $\sigma'(z)$ na saída e a adição do bias na 1ª camada oculta elevaram a AUC-ROC de $0{,}9419$ para $0{,}9780$ e o Recall de $91{,}43\%$ para $96{,}19\%$.
    \item \textbf{Padronização Z-Score via \texttt{StandardScaler} (C1 $\to$ C2):} Conforme demonstrado teoricamente por LeCun et al. \cite{lecun1998efficient}, entradas centradas em média zero ($\mu=0, \sigma=1$) evitam que todos os gradientes $\partial \mathcal{L}/\partial w_{ij} = a_i \delta_j$ compartilhem o mesmo sinal, eliminando oscilações em zigue-zague. Essa mudança reduziu a perda BCE no teste de $0{,}2476$ para $0{,}0010$ e elevou a acurácia no teste de $87{,}32\%$ para $100{,}00\%$.
    \item \textbf{Topologia {[13$\to$16$\to$8$\to$1]} com \texttt{ReLU} e Inicialização de He (C2 $\to$ C4):} Ampliar a 1ª camada oculta de 10 para 16 neurônios eliminou o gargalo de representação dos 13 atributos de entrada, enquanto a ativação \texttt{ReLU} combinada à inicialização de He \cite{he2015delving} ($\sigma_w = \sqrt{2/\text{fan\_in}}$) e $\eta = 0{,}015$ reduziu a perda BCE de teste ao mínimo global de $\mathbf{0{,}0003}$ (redução de $99{,}88\%$ em relação a C1).
\end{enumerate}
